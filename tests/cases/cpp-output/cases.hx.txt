--- property-compound-assign-in-branch + --macro Dump.after('cppout/src/P.cpp')
class W {
	public var left(get, set):Null<Float>;
	var l:Float = 0;
	function get_left():Null<Float> return l;
	function set_left(v:Null<Float>):Null<Float> return l = v;
	public var width(get, set):Null<Float>;
	function get_width():Null<Float> return l;
	function set_width(v:Null<Float>):Null<Float> return l = v;
	public function new() {}
}

class P {
	var w:W = new W();
	var r:Bool = true;

	function cond(d:Float) {
		var prev = w.width;
		if (r) {
			w.width = 3;
		} else if (d > 0) {
			w.width = 4;
			w.left += prev - w.width;
			w.left += d;
		}
	}

	public function new() {}

	static function main() {
		new P().cond(2);
	}
}

--- inline-nullable-arg-value + --macro Dump.after('cppout/src/P.cpp')
class P {
	static inline function either(name:String, other:String):String {
		if (name == "a") return name;
		return other;
	}
	static function f(m:Map<String, Int>, a:Null<String>, b:Null<String>) {
		inline function resolve(name:String):String {
			if (name == "a") return name;
			if (m.exists(name)) return "q" + name;
			throw "unknown " + name;
		}
		var id = resolve(a);
		Sys.println(id);
		var e = either(a, b);
		Sys.println(e);
	}
	static function main() {
		f(["x" => 1], "x", "y");
	}
}

--- macro-sin-cos + --macro Dump.after('cppout/src/P.cpp')
class P {
	macro static function trig() {
		var xs = [Math.PI / 2, Math.PI, 3 * Math.PI / 2, 28.8, 0.5, -2.0, 1e-300, 1e10, -6.5e15, 1e18, 9.3e18, 1e22, -1e300];
		var out = [];
		for (x in xs) out.push(x + " " + Math.cos(x) + " " + Math.sin(x));
		out.push("zero " + (1 / Math.sin(0.0)) + " " + (1 / Math.sin(-0.0)) + " " + Math.cos(-0.0));
		out.push("special " + Math.sin(Math.NaN) + " " + Math.cos(Math.POSITIVE_INFINITY) + " " + Math.sin(Math.NEGATIVE_INFINITY));
		// A NaN comes back as itself, sign and payload kept; an infinity gives the positive quiet NaN.
		var bits = function(f:Float) {
			var b = haxe.io.FPHelper.doubleToI64(f);
			return StringTools.hex(b.high, 8) + StringTools.hex(b.low, 8);
		};
		for (n in [haxe.io.FPHelper.i64ToDouble(0x1234, 0x7FF80001), haxe.io.FPHelper.i64ToDouble(5, 0xFFFC0000)])
			out.push("nan " + bits(n) + " " + bits(Math.sin(n)) + " " + bits(Math.cos(n)));
		out.push("inf " + bits(Math.sin(Math.POSITIVE_INFINITY)) + " " + bits(Math.cos(Math.NEGATIVE_INFINITY)));
		return macro $v{out};
	}
	static function main() {
		var lines:Array<String> = trig();
		for (s in lines) Sys.println(s);
	}
	// Host-dependent: this case's expected output holds only on a Windows host. Haxe's Windows binaries
	// compute sin and cos with the x87 (FSIN/FCOS), and hxc's macro interpreter emulates that (X87Trig) only
	// when hxc.core.Globals.isWindows. Elsewhere Haxe and hxc both call the C library, which differs from these
	// recorded values in the last digits, and the recording also has Windows' three-digit exponents (e-017).
	// hxc's own tests (tests/cases against tests/expected) run on windows-x64 only (.github/workflows/ci.yml).
	// The note sits below main so that the line numbers in the recorded C++ stay as they are.
}
