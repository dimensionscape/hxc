--- hlc-type-names
// Type names (make_types_idents): repeated Null<Int> arguments, a recursive anonymous type, nested anonymous
// types with the same field names, a closure context (t$ctx_) used by one module, and the abstract probe:
// one hl.Abstract used three times through a typedef (one name string in genhl), three separately written
// hints (three strings), and the two mixed.
typedef R = {next:Null<R>, v:Int};
typedef X = hl.Abstract<"hlc_x">;

class P {
	static function nulls(a:Null<Int>, b:Null<Int>):Int {
		return a == null ? (b == null ? 0 : b) : a;
	}

	static function chain(r:R):Int {
		return r.next == null ? r.v : r.v + chain(r.next);
	}

	static function nested(o:{a:{a:Int}}):Int {
		return o.a.a;
	}

	static function direct(a:hl.Abstract<"hlc_d">, b:hl.Abstract<"hlc_d">):hl.Abstract<"hlc_d"> {
		return a;
	}

	static function viaTypedef(a:X, b:X):X {
		return b;
	}

	static function mixedDecl(a:X, b:hl.Abstract<"hlc_x">):Void {}

	static function main() {
		var x = Std.random(5), y = Std.random(3) * 0.5;
		var f = () -> x + y;
		Sys.println(nulls(x, null) + chain({next: {next: null, v: 2}, v: 1}) + nested({a: {a: 3}}) + f());
		direct(null, null);
		viaTypedef(null, null);
		mixedDecl(null, null);
	}
}

--- hlc-dynamic-fields
// Dynamic field reads, writes and calls on Dynamic and on anonymous objects: hl/hashes.c, and the hash
// collision loop (fcbzlgd and fhnmljx have the same hl_hash).
class P {
	static function make():Dynamic {
		return {fcbzlgd: 1, fhnmljx: 2, call: (s:String) -> s + "!"};
	}

	static function main() {
		var d = make();
		d.fcbzlgd = d.fhnmljx + 1;
		d.extra = 3.5;
		Sys.println(d.fcbzlgd + d.extra);
		Sys.println(d.call("hey"));
		var o:{fhnmljx:Int, fcbzlgd:Int, call:String->String} = d;
		o.fhnmljx = o.fcbzlgd * 2;
		Sys.println(o.call("o") + o.fhnmljx);
		var od:Dynamic = {name: "x", f: (i:Int) -> i * 2};
		od.name = "y";
		Sys.println(od.f(od.name.length));
	}

--- hlc-ops
// generate_function arms the HL cases don't reach. Each operation is a function of its own, so the analyzer
// can't fold it into main: integer OSDiv (I32 and I64), OUMod, OUShr (I32 and I64), OXor, OToUFloat, fmod and
// fmodf, OJNotLt and OJNotGte, HNull CNeq, dynamic CGt and CGte, virtual against virtual and against an
// object (both ways round), OVirtualClosure on an object, virtual field reads and ODynGet of F64, F32 and
// I64 (through a type parameter constrained to an anonymous type).
typedef V = {x:Int};

class C {
	public var x:Int;

	public function new(x:Int) {
		this.x = x;
	}

	public function m():Int {
		return x;
	}
}

class D extends C {
	override public function m():Int {
		return x + 1;
	}
}

class P {
	static function sdiv(a:Int, b:Int):Int return Std.int(a / b);
	static function umod(a:UInt, b:UInt):UInt return a % b;
	static function ushr(a:Int, b:Int):Int return a >>> b;
	static function xor(a:Int, b:Int):Int return a ^ b;
	static function ufloat(a:UInt):Float return a;
	static function fmod(a:Float, b:Float):Float return a % b;
	static function fmodf(a:Single, b:Single):Single return a % b;
	static function ldiv(a:hl.I64, b:hl.I64):hl.I64 return a / b;
	static function lushr(a:hl.I64, b:Int):hl.I64 return a >>> b;
	static function notLt(a:Float, b:Float):Int {
		if (a < b) return 1;
		return 0;
	}
	static function notGte(a:Float, b:Float):Int {
		if (a >= b) return 1;
		return 0;
	}
	static function nullNe(a:Null<Int>, b:Null<Int>):Bool return a != b;
	static function dynGt(a:Dynamic, b:Dynamic):Bool return a > b;
	static function dynGte(a:Dynamic, b:Dynamic):Int {
		if (a > b) return 1;
		return 0;
	}
	static function virtEq(a:V, b:V):Bool return a == b;
	static function virtNe(a:V, b:V):Bool return a != b;
	static function virtOther(a:V, b:{x:Int, y:Int}):Bool return a == b;
	static function virtObjEq(a:V, b:C):Bool return a == b;
	static function virtObjNe(a:V, b:C):Bool return a != b;
	static function objVirtEq(a:C, b:V):Bool return a == b;
	static function objVirtNe(a:C, b:V):Bool return a != b;
	static function method(c:C):Void->Int return c.m;
	static function virtFields(o:{f:Float, s:Single, l:hl.I64}):Float return o.f + o.s + o.l.toInt();
	static function getF<T:{p:Float}>(t:T):Float return t.p;
	static function getS<T:{p:Single}>(t:T):Single return t.p;
	static function getL<T:{p:hl.I64}>(t:T):hl.I64 return t.p;

	static function main() {
		var a = Std.random(100) + 1, b = Std.random(7) + 1;
		var fa = a * 0.5, fb = b * 0.25;
		var la:hl.I64 = a, lb:hl.I64 = b;
		var sa:Single = fa, sb:Single = fb;
		var c = a > 50 ? new C(a) : new D(b);
		var v:V = {x: a};
		Sys.println([sdiv(a, b), umod(a, b), ushr(a, b), xor(a, b), ldiv(la, lb).toInt(), lushr(la, b).toInt(), notLt(fa, fb), notGte(fa, fb), dynGte(fa, fb)]);
		Sys.println([ufloat(a), fmod(fa, fb), fmodf(sa, sb), virtFields({f: fa, s: sa, l: la}), getF({p: fa}), getS({p: sa}), getL({p: la}).toInt()]);
		Sys.println([nullNe(a, null), dynGt(fa, fb), virtEq(v, {x: b}), virtNe(v, v), virtOther(v, {x: a, y: b}), virtObjEq(v, c), virtObjNe(v, c), objVirtEq(c, v), objVirtNe(c, v)]);
		Sys.println(method(c)());
	}
}
