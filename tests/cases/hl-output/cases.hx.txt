class P {
	static function find(limit:Int):Int {
		var i = 0;
		while (true) {
			if (i * i > limit) return i;
			i++;
		}
		return -1;
	}

	static function main() {
		var n = 10;
		var total = 0;
		for (i in 0...n) {
			if (i % 2 == 0) continue;
			if (i > 7) break;
			total += i;
		}
		Sys.println(total);
		var j = 0;
		while (j < 5) {
			j++;
		}
		Sys.println(j);
		var k = 0;
		do {
			k += 3;
		} while (k < 10);
		Sys.println(k);
		var count = 0;
		for (a in 0...4) {
			for (b in 0...a) {
				count++;
			}
		}
		Sys.println(count);
		if (count > 5) {
			Sys.println("big");
		} else if (count > 2) {
			Sys.println("medium");
		} else {
			Sys.println("small");
		}
		Sys.println(find(50));
	}
}

--- enums-switch
enum Color {
	Red;
	Green;
	Blue;
}

enum Shape {
	Circle(r:Float);
	Rect(w:Int, h:Int);
	Named(name:String, inner:Shape);
	Empty;
}

enum Solo {
	Only(v:Int);
}

class P {
	static function area(s:Shape):Float {
		return switch s {
			case Circle(r): r * r * 3;
			case Rect(w, h) if (w == h): w * w;
			case Rect(w, h): w * h;
			case Named(_, inner): area(inner);
			case Empty: 0;
		}
	}

	static function colorName(c:Color):String {
		return switch c {
			case Red: "red";
			case Green: "green";
			case Blue: "blue";
		}
	}

	static function bucket(i:Int):String {
		return switch i {
			case 0: "zero";
			case 1 | 2: "small";
			case 10: "ten";
			default: "other";
		}
	}

	static function code(s:String):Int {
		return switch s {
			case "a": 1;
			case "bb": 2;
			case "héllo": 3;
			default: -1;
		}
	}

	static function main() {
		Sys.println(area(Circle(1.0)));
		Sys.println(area(Rect(2, 2)));
		Sys.println(area(Rect(2, 3)));
		Sys.println(area(Named("n", Rect(1, 5))));
		Sys.println(area(Empty));
		Sys.println(colorName(Green));
		Sys.println(Type.enumIndex(Blue));
		Sys.println(Type.enumIndex(Rect(1, 1)));
		for (i in [0, 2, 10, 99]) Sys.println(bucket(i));
		for (s in ["a", "bb", "héllo", "z"]) Sys.println(code(s));
		var c = Red;
		Sys.println(c == Red);
		var o = Only(5);
		switch o {
			case Only(v): Sys.println(v);
		}
		Sys.println(Type.enumConstructor(Rect(1, 2)));
		Sys.println(Type.enumParameters(Named("x", Empty)).length);
	}
}

		return 2;

	public function new() {}
