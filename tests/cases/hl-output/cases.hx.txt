			i++;

		} else if (count > 2) {

		return switch s {
			case Circle(r): r * r * 3;
			case Rect(w, h) if (w == h): w * w;
			case Rect(w, h): w * h;
			case Named(_, inner): area(inner);
			case Empty: 0;
		}

		return switch c {
			case Red: "red";
			case Green: "green";
			case Blue: "blue";

		return switch i {
			case 0: "zero";
			case 1 | 2: "small";
			case 10: "ten";

		}
	}

	static function code(s:String):Int {

class MyException extends haxe.Exception {
	public var code:Int;

	public function new(msg:String, code:Int) {
		super(msg);
		this.code = code;
	}
}

class P {
	static function fail(kind:Int):Void {
		switch kind {
			case 0: throw "plain";
			case 1: throw new MyException("mine", 7);
			case 2: throw 42;
			case -1: throw "negative";
			case 1000: throw new haxe.Exception("big");
			case _:
		}
	}

	static function attempt(kind:Int):String {
		try {
			fail(kind);
			return "none";
		} catch (e:String) {
			return "string " + e;
		} catch (e:MyException) {
			return "mine " + e.message + " " + e.code;
		} catch (e:Dynamic) {
			return "dynamic " + Std.string(e);
		}
	}

	static function rethrow():Void {
		try {
			fail(1);
		} catch (e:MyException) {
			Sys.println("caught once: " + e.message);
			throw e;
		}
	}

	static function main() {
		for (k in [0, 1, 2, 3, -1, 1000]) Sys.println(attempt(k));
		try {
			rethrow();
		} catch (e:haxe.Exception) {
			Sys.println("caught again: " + e.message + " " + (e.stack != null));
		}
		var count = 0;
		for (i in 0...10) {
			try {
				if (i == 1) continue;
				if (i == 5) break;
				count += i;
			} catch (e:Dynamic) {
				count = -1;
			}
		}
		Sys.println(count);
		var v = try Std.parseInt("12") catch (e:Dynamic) -1;
		Sys.println(v);
		var w:Int = try {
			fail(2);
			0;
		} catch (e:Dynamic) {
			5;
		}
		Sys.println(w);
	}
}

@:generic
class Stack<T> {
	var items:Array<T> = [];

	public function new() {}

	public function push(v:T):Void {
		items.push(v);
	}

	public function pop():T {
		return items.pop();
	}

	public var size(get, never):Int;

	function get_size():Int {
		return items.length;
	}
}

abstract Meters(Int) {
	public inline function new(v:Int) {
		this = v;
	}

	@:from static function fromString(s:String):Meters {
		return new Meters(s.length);
	}

	@:to function toFloat():Float {
		return this / 2;
	}

	@:op(A + B) static function add(a:Meters, b:Meters):Meters;

	@:op(A * B) static function scale(a:Meters, b:Int):Meters {
		return new Meters(a.raw() * b);
	}

	@:op(-A) static function neg(a:Meters):Meters;

	@:op(A > B) static function gt(a:Meters, b:Meters):Bool;

	@:op(A == B) static function eq(a:Meters, b:Meters):Bool;

	public inline function raw():Int {
		return this;
	}

	public function toString():String {
		return this + "m";
	}
}

# A recursive generic typedef: each apply_typedef makes a fresh anon, and the second visit of Node<Int> finds
# the first in genhl's anons_cache only because OCaml compares the anons structurally. Pair swaps its
# parameters, so the hit is two applications away; Tree recurses through Array<>. No second instance of Node:
# Haxe 4.3.7 itself runs out of memory comparing Node<Int>'s virtual with Node<String>'s (the recursive field
# `next` sorts before `value`, so OCaml's compare recurses through `next` without end).
--- recursive-typedef
typedef Node<T> = {value:T, ?next:Node<T>}
typedef Pair<A, B> = {a:A, b:Null<Pair<B, A>>}
typedef Tree<T> = {label:T, children:Array<Tree<T>>}

class P {
	static function sum(n:Node<Int>):Int {
		var s = 0;
		while (n != null) {
			s += n.value;
			n = n.next;
		}
		return s;
	}

	static function count<T>(t:Tree<T>):Int {
		var c = 1;
		for (x in t.children) c += count(x);
		return c;
	}

	static function main() {
		var n:Node<Int> = {value: 1, next: {value: 2}};
		Sys.println(n.next.value);
		Sys.println(sum({value: 3, next: n}));
		var p:Pair<Int, String> = {a: 1, b: {a: "x", b: {a: 2, b: null}}};
		Sys.println(p.b.a + p.b.b.a);
		var t:Tree<String> = {label: "root", children: [{label: "leaf", children: []}, {label: "leaf2", children: []}]};
		Sys.println(t.children[1].label + " " + count(t));
	}
}
