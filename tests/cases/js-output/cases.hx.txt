--- an-inline-iterable-argument-keeps-a-map-fields-dynamic-iterator
using Ext;
class P {
	var sm:Map<String, String> = [];
	var im:Map<Int, String> = [];
	function new() {}
	static inline function first(it:Iterable<String>):Bool return it.iterator().hasNext();
	function run() {
		for (c in sm.notNull()) trace(c);
		for (c in im.notNull()) trace(c);
		trace(first(sm));
		trace(Lambda.map(sm, (c) -> c == null));
	}
	static function main() { new P().run(); }
}

--- an-empty-if-on-a-reassigned-local-keeps-the-local + -D analyzer-optimize
class P {
	var target:Int = 2;
	function new() {}
	function a() {
		var s = switch (target) { case 2: "a"; case 3: "b"; case _: null; }
		if (null != s) {}
	}
	function b() {
		var s = null;
		if (target == 2) s = "a";
		if (null != s) {}
	}
	function c(q:String) {
		var s = target;
		if (3 != s) {}
		if (null != q) {}
		q;
	}
	static function main() { var p = new P(); p.a(); p.b(); p.c("x"); }
}
