--- typed-dump + --hl out.hl -D hl_no_opt -D dump -w -WGenerator --macro Probe.after('dump/hl/P.dump')
enum E {
	A;
	B(n:Int);
}

class C {
	public var x:Int;
	public function new(x:Int) this.x = x;
	public function get():Int return x;
}

class P {
	static function f(e:E):Int {
		return switch (e) {
			case A: 0;
			case B(n): n;
		}
	}
	static function main() {
		var c = new C(3);
		var g = function(y:Int) return y + c.get();
		var r = 0;
		try {
			r = g(f(B(2)));
			if (r > 100) throw "big";
		} catch (s:String) {
			r = -1;
		}
		trace(r);
	}
}

--- typed-dump-postfix-property + --hl out.hl -D hl_no_opt -D dump -w -WGenerator --macro Probe.after('dump/hl/P.dump')
class Box {
	public var prop(get, set):Int;
	public static var sprop(default, set):Int = 3;

	var store:Int = 2;

	public function new() {}

	function get_prop() {
		return store;
	}

	function set_prop(x:Int) {
		return store = x;
	}

	static function set_sprop(x:Int) {
		return sprop = x;
	}
}

class P {
	static function main() {
		var b = new Box();
		Sys.println(b.prop++);
		Sys.println(Box.sprop++);
		b.prop++;
	}
}
